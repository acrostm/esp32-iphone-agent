cmake_minimum_required(VERSION 3.16)
set(BOARD "m5stack-stopwatch" CACHE STRING "Supported board ID")
if(NOT BOARD STREQUAL "m5stack-stopwatch")
    message(FATAL_ERROR "Unsupported BOARD: ${BOARD}. Add a board profile and driver before building.")
endif()
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/boards/${BOARD}/sdkconfig.defaults")
set(M5UNIFIED_ARDUINO_COMPONENT OFF CACHE STRING "" FORCE)
set(M5GFX_ARDUINO_COMPONENT OFF CACHE STRING "" FORCE)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(stopwatch_agent)
